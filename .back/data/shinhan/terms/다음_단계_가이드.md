### 1.3 적재 스크립트 (load_to_postgres.py)

```python
"""
신한카드 약관 데이터 PostgreSQL 적재 스크립트
"""

import json
import psycopg2
from psycopg2.extras import execute_batch
from pathlib import Path
from datetime import datetime

# 경로 설정
BASE_DIR = Path(__file__).resolve().parents[2]
DATA_FILE = BASE_DIR / "data" / "sinhan_terms" / "final" / "sinhan_terms_for_rdb.json"  # 최종 988개 버전

# PostgreSQL 연결 설정
DB_CONFIG = {
    "host": "localhost",
    "port": 5432,
    "database": "call_act_db",
    "user": "your_username",
    "password": "your_password"
}


def load_json_data(file_path: Path):
    """JSON 파일 로드"""
    with open(file_path, 'r', encoding='utf-8') as f:
        return json.load(f)


def insert_documents(conn, documents):
    """
    배치 삽입 (execute_batch 사용)
    """
    cursor = conn.cursor()

    insert_query = """
    INSERT INTO service_guide_documents
        (id, title, content, metadata, source_type, created_at, updated_at)
    VALUES
        (%s, %s, %s, %s, %s, %s, %s)
    ON CONFLICT (id)
    DO UPDATE SET
        title = EXCLUDED.title,
        content = EXCLUDED.content,
        metadata = EXCLUDED.metadata,
        updated_at = EXCLUDED.updated_at
    """

    # 데이터 준비
    data_tuples = []
    now = datetime.now()

    for doc in documents:
        data_tuples.append((
            doc["id"],
            doc["title"],
            doc["content"],
            json.dumps(doc["metadata"], ensure_ascii=False),  # JSONB로 변환
            doc["source_type"],
            now,
            now
        ))

    # 배치 삽입 (페이지 크기: 100)
    execute_batch(cursor, insert_query, data_tuples, page_size=100)

    conn.commit()
    cursor.close()

    return len(data_tuples)


def verify_insertion(conn, source_type="sinhan_terms"):
    """삽입 검증"""
    cursor = conn.cursor()

    # 개수 확인
    cursor.execute(
        "SELECT COUNT(*) FROM service_guide_documents WHERE source_type = %s",
        (source_type,)
    )
    count = cursor.fetchone()[0]

    # 샘플 조회
    cursor.execute(
        """
        SELECT id, title, metadata->>'category1' as category1
        FROM service_guide_documents
        WHERE source_type = %s
        LIMIT 5
        """,
        (source_type,)
    )
    samples = cursor.fetchall()

    cursor.close()

    return count, samples


def main():
    """메인 실행 함수"""
    print("[INFO] RDB 적재 시작\n")

    # 1. JSON 파일 로드
    print(f"[1/4] JSON 파일 로드 중...")
    documents = load_json_data(DATA_FILE)
    print(f"  - 로드된 문서 수: {len(documents)}\n")

    # 2. PostgreSQL 연결
    print(f"[2/4] PostgreSQL 연결 중...")
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        print(f"  - 연결 성공: {DB_CONFIG['database']}\n")
    except Exception as e:
        print(f"  [ERROR] 연결 실패: {e}")
        return

    # 3. 데이터 삽입
    print(f"[3/4] 데이터 삽입 중...")
    try:
        inserted_count = insert_documents(conn, documents)
        print(f"  - 삽입 완료: {inserted_count}개\n")
    except Exception as e:
        print(f"  [ERROR] 삽입 실패: {e}")
        conn.rollback()
        conn.close()
        return

    # 4. 검증
    print(f"[4/4] 삽입 검증 중...")
    count, samples = verify_insertion(conn)
    print(f"  - DB 내 문서 수: {count}개")
    print(f"  - 샘플 (5개):")
    for sample in samples:
        print(f"    - {sample[0]}: {sample[1]} (category1: {sample[2]})")

    # 연결 종료
    conn.close()

    print("\n[SUCCESS] RDB 적재 완료!")


if __name__ == "__main__":
    main()
```

## 2. 임베딩 생성 (OpenAI API)

### 2.1 목표

`sinhan_terms_for_vectordb.json` 파일의 988개 청크를 OpenAI text-embedding-3-small 모델로 임베딩 생성

### 2.2 비용 계산

**모델**: `text-embedding-3-small`
- **가격**: $0.02 / 1M tokens
- **예상 토큰 수**: 988개 청크 × 평균 465자 × 1.3 (한글 토큰 비율) ≈ 596,000 tokens
- **예상 비용**: $0.01~0.02 (약 15~25원)

**참고**: `text-embedding-3-large`는 더 높은 성능이지만 비용도 높음 ($0.13 / 1M tokens)

### 2.3 임베딩 생성 스크립트 (generate_embeddings.py)

```python
"""
신한카드 약관 데이터 임베딩 생성 스크립트
OpenAI text-embedding-3-small 사용
"""

import json
import time
from pathlib import Path
from typing import List, Dict
from openai import OpenAI
from tqdm import tqdm

# 경로 설정
BASE_DIR = Path(__file__).resolve().parents[2]
INPUT_FILE = BASE_DIR / "data" / "sinhan_terms" / "final" / "sinhan_terms_for_vectordb.json"  # 최종 988개
OUTPUT_FILE = BASE_DIR / "data" / "sinhan_terms" / "final" / "sinhan_terms_with_embeddings.json"  # 임베딩 포함

# OpenAI 설정
OPENAI_API_KEY = "your_openai_api_key_here"  # 환경 변수 권장
EMBEDDING_MODEL = "text-embedding-3-small"
EMBEDDING_DIMENSION = 1536
BATCH_SIZE = 100  # OpenAI API는 배치당 최대 2048개


def load_documents(file_path: Path) -> List[Dict]:
    """문서 로드"""
    with open(file_path, 'r', encoding='utf-8') as f:
        return json.load(f)


def generate_embeddings_batch(client: OpenAI, texts: List[str]) -> List[List[float]]:
    """
    배치 임베딩 생성
    """
    try:
        response = client.embeddings.create(
            model=EMBEDDING_MODEL,
            input=texts,
            encoding_format="float"
        )

        # 임베딩 추출
        embeddings = [item.embedding for item in response.data]

        return embeddings

    except Exception as e:
        print(f"[ERROR] 임베딩 생성 실패: {e}")
        raise


def process_documents_with_retry(client: OpenAI, documents: List[Dict], batch_size: int = BATCH_SIZE):
    """
    문서 배치 처리 (재시도 로직 포함)
    """
    total = len(documents)
    results = []

    print(f"[INFO] 총 {total}개 문서 처리 시작 (배치 크기: {batch_size})\n")

    for i in tqdm(range(0, total, batch_size), desc="임베딩 생성"):
        batch = documents[i:i + batch_size]
        texts = [doc["text"] for doc in batch]

        # 재시도 로직 (최대 3회)
        for attempt in range(3):
            try:
                embeddings = generate_embeddings_batch(client, texts)

                # 결과 저장
                for doc, embedding in zip(batch, embeddings):
                    results.append({
                        "id": doc["id"],
                        "text": doc["text"],
                        "embedding": embedding,
                        "metadata": doc["metadata"]
                    })

                # 성공 시 다음 배치로
                break

            except Exception as e:
                if attempt < 2:
                    wait_time = (attempt + 1) * 5
                    print(f"\n[RETRY] {wait_time}초 후 재시도... (시도 {attempt + 2}/3)")
                    time.sleep(wait_time)
                else:
                    print(f"\n[ERROR] 배치 {i//batch_size + 1} 실패: {e}")
                    raise

        # Rate limit 방지 (OpenAI 무료 플랜: 3 RPM)
        time.sleep(0.5)

    return results


def save_embeddings(documents: List[Dict], output_file: Path):
    """임베딩 결과 저장"""
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(documents, f, ensure_ascii=False, indent=2)


def verify_embeddings(documents: List[Dict]):
    """임베딩 검증"""
    print("\n[검증]")
    print(f"  - 총 문서 수: {len(documents)}")

    if documents:
        first_embedding = documents[0]["embedding"]
        print(f"  - 임베딩 차원: {len(first_embedding)}")
        print(f"  - 샘플 임베딩 (처음 5개 값): {first_embedding[:5]}")

        # 모든 임베딩 차원 확인
        dimensions = [len(doc["embedding"]) for doc in documents]
        if len(set(dimensions)) == 1:
            print(f"  - 모든 임베딩 차원 일치: {dimensions[0]}")
        else:
            print(f"  [WARNING] 임베딩 차원 불일치 발견!")


def estimate_cost(total_tokens: int):
    """비용 추정"""
    cost_per_million = 0.02  # text-embedding-3-small
    estimated_cost = (total_tokens / 1_000_000) * cost_per_million
    print(f"\n[비용 추정]")
    print(f"  - 예상 토큰 수: {total_tokens:,}")
    print(f"  - 예상 비용: ${estimated_cost:.4f}")


def main():
    """메인 실행 함수"""
    print("[INFO] 임베딩 생성 시작\n")

    # 1. 문서 로드
    print("[1/5] 문서 로드 중...")
    documents = load_documents(INPUT_FILE)
    print(f"  - 로드된 문서 수: {len(documents)}\n")

    # 2. 비용 추정
    total_chars = sum(len(doc["text"]) for doc in documents)
    estimated_tokens = int(total_chars * 1.3)  # 한글 토큰 비율
    estimate_cost(estimated_tokens)

    # 사용자 확인
    print("\n계속 진행하시겠습니까? (y/n): ", end="")
    # 실제 사용 시 주석 해제
    # if input().lower() != 'y':
    #     print("작업 취소")
    #     return

    # 3. OpenAI 클라이언트 초기화
    print("\n[2/5] OpenAI 클라이언트 초기화 중...")
    client = OpenAI(api_key=OPENAI_API_KEY)
    print("  - 초기화 완료\n")

    # 4. 임베딩 생성
    print("[3/5] 임베딩 생성 중...")
    try:
        results = process_documents_with_retry(client, documents)
        print(f"\n  - 생성 완료: {len(results)}개\n")
    except Exception as e:
        print(f"\n[ERROR] 임베딩 생성 실패: {e}")
        return

    # 5. 검증
    print("[4/5] 검증 중...")
    verify_embeddings(results)

    # 6. 저장
    print("\n[5/5] 저장 중...")
    save_embeddings(results, OUTPUT_FILE)
    file_size = OUTPUT_FILE.stat().st_size / (1024 * 1024)
    print(f"  - 저장 완료: {OUTPUT_FILE}")
    print(f"  - 파일 크기: {file_size:.1f}MB")

    print("\n[SUCCESS] 임베딩 생성 완료!")


if __name__ == "__main__":
    main()
```

### 3.3 VectorDB 적재 스크립트 (load_to_vectordb.py)

```python
"""
신한카드 약관 임베딩 VectorDB 적재 스크립트
PostgreSQL pgvector 사용
"""

import json
import psycopg2
from psycopg2.extras import execute_batch
from pathlib import Path

# 경로 설정
BASE_DIR = Path(__file__).resolve().parents[2]
EMBEDDING_FILE = BASE_DIR / "data" / "sinhan_terms" / "final" / "sinhan_terms_with_embeddings.json"

# PostgreSQL 연결 설정
DB_CONFIG = {
    "host": "localhost",
    "port": 5432,
    "database": "call_act_db",
    "user": "your_username",
    "password": "your_password"
}


def load_embeddings(file_path: Path):
    """임베딩 JSON 로드"""
    with open(file_path, 'r', encoding='utf-8') as f:
        return json.load(f)


def update_embeddings(conn, documents):
    """
    임베딩 업데이트 (embedding 컬럼만)
    """
    cursor = conn.cursor()

    update_query = """
    UPDATE service_guide_documents
    SET
        embedding = %s::vector,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = %s
    """

    # 데이터 준비
    data_tuples = []
    for doc in documents:
        embedding_str = "[" + ",".join(map(str, doc["embedding"])) + "]"
        data_tuples.append((embedding_str, doc["id"]))

    # 배치 업데이트
    execute_batch(cursor, update_query, data_tuples, page_size=100)

    conn.commit()
    cursor.close()

    return len(data_tuples)


def verify_embeddings(conn):
    """임베딩 저장 검증"""
    cursor = conn.cursor()

    # 1. 임베딩 개수 확인
    cursor.execute("""
        SELECT COUNT(*)
        FROM service_guide_documents
        WHERE source_type = 'sinhan_terms'
        AND embedding IS NOT NULL
    """)
    count = cursor.fetchone()[0]

    # 2. 샘플 유사도 검색 테스트
    cursor.execute("""
        SELECT id, title, embedding
        FROM service_guide_documents
        WHERE source_type = 'sinhan_terms'
        AND embedding IS NOT NULL
        LIMIT 1
    """)
    sample = cursor.fetchone()

    if sample:
        sample_id, sample_title, sample_embedding = sample

        # 자기 자신과 가장 유사한 문서 5개 검색
        cursor.execute("""
            SELECT
                id,
                title,
                1 - (embedding <=> %s::vector) as similarity
            FROM service_guide_documents
            WHERE source_type = 'sinhan_terms'
            AND embedding IS NOT NULL
            ORDER BY embedding <=> %s::vector
            LIMIT 5
        """, (str(sample_embedding), str(sample_embedding)))

        similar_docs = cursor.fetchall()
    else:
        similar_docs = []

    cursor.close()

    return count, sample_title if sample else None, similar_docs


def create_vector_index(conn):
    """벡터 인덱스 생성"""
    cursor = conn.cursor()

    print("\n[INFO] 벡터 인덱스 생성 중... (시간이 걸릴 수 있습니다)")

    # HNSW 인덱스 생성
    cursor.execute("""
        CREATE INDEX IF NOT EXISTS idx_embedding_hnsw
        ON service_guide_documents
        USING hnsw (embedding vector_cosine_ops)
    """)

    conn.commit()
    cursor.close()

    print("  - 인덱스 생성 완료")


def main():
    """메인 실행 함수"""
    print("[INFO] VectorDB 적재 시작\n")

    # 1. 임베딩 파일 로드
    print("[1/5] 임베딩 파일 로드 중...")
    documents = load_embeddings(EMBEDDING_FILE)
    print(f"  - 로드된 문서 수: {len(documents)}")

    if documents:
        print(f"  - 임베딩 차원: {len(documents[0]['embedding'])}\n")

    # 2. PostgreSQL 연결
    print("[2/5] PostgreSQL 연결 중...")
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        print(f"  - 연결 성공: {DB_CONFIG['database']}\n")
    except Exception as e:
        print(f"  [ERROR] 연결 실패: {e}")
        return

    # 3. 임베딩 업데이트
    print("[3/5] 임베딩 업데이트 중...")
    try:
        updated_count = update_embeddings(conn, documents)
        print(f"  - 업데이트 완료: {updated_count}개\n")
    except Exception as e:
        print(f"  [ERROR] 업데이트 실패: {e}")
        conn.rollback()
        conn.close()
        return

    # 4. 검증
    print("[4/5] 임베딩 검증 중...")
    count, sample_title, similar_docs = verify_embeddings(conn)
    print(f"  - 임베딩 저장 개수: {count}")

    if similar_docs:
        print(f"\n  - 유사도 검색 테스트 (기준: {sample_title}):")
        for doc_id, title, similarity in similar_docs:
            print(f"    - {title}: {similarity:.4f}")

    # 5. 벡터 인덱스 생성
    print("\n[5/5] 벡터 인덱스 생성 중...")
    try:
        create_vector_index(conn)
    except Exception as e:
        print(f"  [WARNING] 인덱스 생성 실패: {e}")
        print("  - 수동으로 인덱스를 생성하세요.")

    # 연결 종료
    conn.close()

    print("\n[SUCCESS] VectorDB 적재 완료!")


if __name__ == "__main__":
    main()
```

### 3.5 유사도 검색 쿼리 예시

```sql
-- 1. 기본 유사도 검색 (코사인 유사도)
-- 특정 임베딩과 가장 유사한 문서 5개 검색

FROM service_guide_documents

-- 2. 카테고리 필터링 + 유사도 검색
SELECT
    id,
    title,
    metadata->>'category2' as category2,
    1 - (embedding <=> %s::vector) as similarity
FROM service_guide_documents
WHERE source_type = 'sinhan_terms'
AND metadata->>'category1' = '신용카드'
ORDER BY embedding <=> %s::vector
LIMIT 10;

-- 3. 유사도 임계값 설정
SELECT
    id,
    title,
    1 - (embedding <=> %s::vector) as similarity
FROM service_guide_documents
WHERE source_type = 'sinhan_terms'
AND 1 - (embedding <=> %s::vector) > 0.7  -- 유사도 70% 이상만
ORDER BY embedding <=> %s::vector
LIMIT 10;

-- 4. 하이브리드 검색 (키워드 + 벡터)
SELECT
    id,
    title,
    content,
    1 - (embedding <=> %s::vector) as similarity,
    ts_rank(to_tsvector('korean', content), plainto_tsquery('korean', '연회비')) as keyword_score
FROM service_guide_documents
WHERE source_type = 'sinhan_terms'
AND to_tsvector('korean', content) @@ plainto_tsquery('korean', '연회비')
ORDER BY
    (1 - (embedding <=> %s::vector)) * 0.7 +  -- 벡터 유사도 70%
    ts_rank(to_tsvector('korean', content), plainto_tsquery('korean', '연회비')) * 0.3  -- 키워드 30%
    DESC
LIMIT 10;
```
