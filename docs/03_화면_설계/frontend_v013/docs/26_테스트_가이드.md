## 1. 테스트 전략

### 1.1 현재 테스트 접근 방식

**CALL:ACT MVP는 현재 수동 테스트 기반으로 개발되고 있습니다.**

| 테스트 유형 | 현재 상태 | 향후 계획 |
|------------|----------|-----------|
| 수동 기능 테스트 | ✅ 진행 중 | 지속 |
| 브라우저 호환성 테스트 | ✅ 진행 중 | 지속 |
| 반응형 테스트 | ✅ 진행 중 | 지속 |
| Unit 테스트 | ❌ 미적용 | Phase 6에서 도입 예정 |
| Integration 테스트 | ❌ 미적용 | Phase 6에서 도입 예정 |
| E2E 테스트 | ❌ 미적용 | Phase 7에서 도입 예정 |

### 1.2 테스트 우선순위

```
1순위: 핵심 기능 (로그인, 실시간 상담, 칸반보드)
2순위: 주요 페이지 (대시보드, 상담 이력, ACW)
3순위: 관리자 기능 (사원 관리, 통계, 공지사항)
4순위: 부가 기능 (프로필, 시뮬레이션)
```

### 3.3 Integration 테스트

**테스트 예시** (참고용):

```typescript
// src/app/pages/__tests__/LoginPage.integration.test.tsx
import { describe, it, expect } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import { BrowserRouter } from 'react-router-dom';
import LoginPage from '../LoginPage';

describe('LoginPage Integration', () => {
  it('should login successfully with valid credentials', async () => {
    render(
      <BrowserRouter>
        <LoginPage />
      </BrowserRouter>
    );

    // 사번 입력
    const employeeIdInput = screen.getByPlaceholderText('사번을 입력하세요');
    fireEvent.change(employeeIdInput, { target: { value: 'EMP-001' } });

    // 비밀번호 입력
    const passwordInput = screen.getByPlaceholderText('비밀번호를 입력하세요');
    fireEvent.change(passwordInput, { target: { value: '0000' } });

    // 로그인 버튼 클릭
    const loginButton = screen.getByRole('button', { name: '로그인' });
    fireEvent.click(loginButton);

    // LocalStorage 확인
    expect(localStorage.getItem('employeeId')).toBe('EMP-001');
    expect(localStorage.getItem('userRole')).toBe('employee');
  });

  it('should show error with invalid credentials', async () => {
    render(
      <BrowserRouter>
        <LoginPage />
      </BrowserRouter>
    );

    fireEvent.change(screen.getByPlaceholderText('사번을 입력하세요'), {
      target: { value: 'INVALID' }
    });
    fireEvent.change(screen.getByPlaceholderText('비밀번호를 입력하세요'), {
      target: { value: 'wrong' }
    });
    fireEvent.click(screen.getByRole('button', { name: '로그인' }));

    // 에러 메시지 표시 확인
    expect(await screen.findByText(/입력된 정보가 올바르지 않습니다/)).toBeInTheDocument();
  });
});
```

// tests/e2e/consultation-flow.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Consultation Flow', () => {
  test('should complete full consultation flow', async ({ page }) => {
    // 1. 로그인
    await page.goto('/login');
    await page.fill('input[type="text"]', 'EMP-001');
    await page.fill('input[type="password"]', '0000');
    await page.click('button:has-text("로그인")');

    // 2. 대시보드 확인
    await expect(page).toHaveURL('/dashboard');
    await expect(page.locator('h1')).toContainText('대시보드');

    // 3. 실시간 상담 페이지 이동
    await page.click('a[href="/consultation/live"]');
    await expect(page).toHaveURL('/consultation/live');

    // 4. 대기 콜 선택
    await page.click('text=카드분실');
    await expect(page.locator('.customer-info')).toContainText('홍길동');

    // 5. 상담 시작
    await page.click('button:has-text("상담 시작")');
    await page.waitForSelector('.stt-message');

    // 6. 카드 자세히 보기
    await page.click('.kanban-card').first();
    await expect(page.locator('.modal')).toBeVisible();
    await expect(page.locator('.modal')).toContainText('제34조');

    // 7. 상담 종료
    await page.click('button:has-text("상담 종료")');
    await page.click('button:has-text("확인")');

    // 8. ACW 페이지 확인
    await expect(page).toHaveURL('/acw');
  });
});
