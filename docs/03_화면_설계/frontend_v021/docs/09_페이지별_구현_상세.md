#### 2.2.3 주요 기능

**1. 로그인 처리**
```typescript
const handleLogin = (e: React.FormEvent) => {
  e.preventDefault();
  
  // LocalStorage에서 사원 목록 확인
  const savedEmployees = localStorage.getItem('employees');
  let employees = savedEmployees ? JSON.parse(savedEmployees) : employeesData;
  
  // 관리자 계정 확인
  if (employeeId === 'ADMIN-001' && password === '0000') {
    localStorage.setItem('userRole', 'admin');
    localStorage.setItem('employeeId', 'ADMIN-001');
    localStorage.setItem('employeeName', '관리자');
    localStorage.setItem('isAdmin', 'true');
    navigate('/dashboard');
    return;
  }
  
  // 사원 계정 확인
  const foundEmployee = employees.find(emp => emp.id === employeeId);
  
  if (foundEmployee && password === '0000') {
    const isAdmin = ['팀장', '부장', '이사'].includes(foundEmployee.position);
    
    localStorage.setItem('userRole', isAdmin ? 'admin' : 'employee');
    localStorage.setItem('employeeId', foundEmployee.id);
    localStorage.setItem('employeeName', foundEmployee.name);
    localStorage.setItem('isAdmin', isAdmin.toString());
    navigate('/dashboard');
  } else {
    setError('입력된 정보가 올바르지 않습니다. (기본 비밀번호: 0000)');
  }
};
```

**2. 데모 계정**
| 사번 | 비밀번호 | 역할 | 설명 |
|------|----------|------|------|
| ADMIN-001 | 0000 | 관리자 | 최고 관리자 |
| EMP-001 | 0000 | 상담사 | 일반 사원 (김민수) |
| EMP-002 | 0000 | 상담사 | 일반 사원 (이영희) |
| EMP-003 | 0000 | 상담사 | 일반 사원 (박철수) |

**참고**: employeesData의 모든 사원은 기본 비밀번호 `0000`으로 로그인 가능

**3. 인증 상태 관리**
- LocalStorage에 저장되는 정보:
  - `userRole`: 'admin' 또는 'employee'
  - `employeeId`: 사번 (예: EMP-001)
  - `employeeName`: 이름
  - `isAdmin`: 'true' 또는 'false'
  - `userTeam`: 소속팀
  - `userPosition`: 직급
  - `userEmail`: 이메일
  - `userPhone`: 전화번호
  - `userJoinDate`: 입사일

**5. Step 자동 전환**
```typescript
// 현재 Step의 모든 키워드가 태깅되면 다음 Step으로
useEffect(() => {
  if (!currentStep) return;
  
  const stepKeywords = currentStep.keywords.map(k => k.text);
  const taggedKeywords = keywords.map(k => k.text);
  
  const allTagged = stepKeywords.every(k => taggedKeywords.includes(k));
  
  if (allTagged && currentStepIndex < 2) {
    // 2초 후 자동 전환 (애니메이션 효과)
    setTimeout(() => {
      setCurrentStepIndex(prev => prev + 1);
    }, 2000);
  }
}, [keywords, currentStep]);
```

```typescript
const filteredConsultations = consultationsData.filter(c => {
  // 카테고리 필터
  if (categoryFilter !== '전체' && c.category !== categoryFilter) {
    return false;
  }
  
  // 상태 필터
  if (statusFilter !== '전체' && c.status !== statusFilter) {
    return false;
  }
  
  // 검색어 필터
  if (searchQuery) {
    const query = searchQuery.toLowerCase();
    return (
      c.id.toLowerCase().includes(query) ||
      c.customer.toLowerCase().includes(query) ||
      c.agent.toLowerCase().includes(query)
    );
  }
  
  return true;
});
```
